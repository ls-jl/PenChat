cmake_minimum_required(VERSION 3.16)
project(penchat_falcon_submit_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(COMPOSITOR "${ROOT}/native/compositor")
add_library(composite_transport STATIC "${COMPOSITOR}/composite_wire.cpp")
target_include_directories(composite_transport PUBLIC "${COMPOSITOR}")
set_target_properties(composite_transport PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(composite_transport PRIVATE -Wall -Wextra)
add_library(submit_probe_core STATIC "${COMPOSITOR}/elf_slot.cpp" probe.cpp "${COMPOSITOR}/composite.cpp")
target_link_libraries(submit_probe_core PUBLIC composite_transport)
set_target_properties(submit_probe_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(submit_probe_core PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
if(CMAKE_CROSSCOMPILING)
  target_include_directories(submit_probe_core PRIVATE "${ROOT}/.deps/openssl-headers-cv1826/include")
  target_link_libraries(submit_probe_core PUBLIC "${ROOT}/.deps/http-system-cv1826/lib/libcrypto.so.1.1" dl pthread)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(submit_probe_core PUBLIC OpenSSL::Crypto dl pthread)
endif()
add_library(submit_probe_fixture SHARED fixture.cpp)
target_include_directories(submit_probe_fixture PRIVATE "${COMPOSITOR}")
target_link_options(submit_probe_fixture PRIVATE -Wl,-z,now,-z,relro)
add_executable(submit_probe_tests test.cpp)
target_compile_options(submit_probe_tests PRIVATE -UNDEBUG)
target_link_libraries(submit_probe_tests PRIVATE submit_probe_core)
add_executable(composite_fixture composite_fixture.cpp)
target_link_libraries(composite_fixture PRIVATE composite_transport pthread)
add_executable(composite_tests composite_test.cpp)
target_compile_options(composite_tests PRIVATE -UNDEBUG)
target_link_libraries(composite_tests PRIVATE submit_probe_core)
add_executable(surface_fixture surface_fixture.cpp "${COMPOSITOR}/surface.cpp")
target_include_directories(surface_fixture PRIVATE "${ROOT}/native/camera/src")
target_link_libraries(surface_fixture PRIVATE composite_transport pthread)
target_compile_options(surface_fixture PRIVATE -UNDEBUG -Wall -Wextra)
add_executable(surface_tests surface_test.cpp)
target_compile_options(surface_tests PRIVATE -UNDEBUG)
target_link_libraries(surface_tests PRIVATE submit_probe_core)
enable_testing()
add_test(NAME submit-probe COMMAND submit_probe_tests $<TARGET_FILE:submit_probe_fixture>)
add_test(NAME composite-probe COMMAND composite_tests $<TARGET_FILE:submit_probe_fixture> $<TARGET_FILE:composite_fixture>)
add_test(NAME composite-surface COMMAND surface_tests $<TARGET_FILE:submit_probe_fixture> $<TARGET_FILE:surface_fixture>)
if(CMAKE_CROSSCOMPILING)
  target_link_libraries(surface_fixture PRIVATE -static-libstdc++ -static-libgcc)
  target_link_libraries(surface_tests PRIVATE -static-libstdc++ -static-libgcc)
  target_compile_options(composite_transport PRIVATE -march=armv7-a -mfpu=neon -mfloat-abi=hard)
  target_link_libraries(composite_fixture PRIVATE -static-libstdc++ -static-libgcc)
  target_link_libraries(composite_tests PRIVATE -static-libstdc++ -static-libgcc)
  add_executable(penchat-composite-client composite_client.cpp
    "${ROOT}/native/lvgl-app/src/native_touch.cpp" "${ROOT}/native/lvgl-app/src/common.cpp")
  target_include_directories(penchat-composite-client PRIVATE "${ROOT}/.deps/lvgl"
    "${ROOT}/native/lvgl-app/src" "${ROOT}/build/cv1826/generated" "${ROOT}/.deps/json/single_include")
  target_compile_definitions(penchat-composite-client PRIVATE LV_CONF_PATH="${ROOT}/native/lvgl-app/lv_conf.h" LV_KCONFIG_IGNORE)
  target_link_libraries(penchat-composite-client PRIVATE composite_transport pthread "${ROOT}/build/cv1826/lvgl/liblvgl.a" m -static-libstdc++ -static-libgcc)
  set_target_properties(penchat-composite-client PROPERTIES SKIP_BUILD_RPATH ON)
  target_link_libraries(submit_probe_tests PRIVATE -static-libstdc++ -static-libgcc)
  if(NOT PENCHAT_SUBMIT_PROBE_APP_ID MATCHES "^800[0-9]+$")
    message(FATAL_ERROR "An isolated probe AppID is required")
  endif()
  file(GLOB SDK_SRC "${ROOT}/native/vendor/iot-miniapp-sdk/src/jqutil_v2/*.cpp")
  add_library(jsapi_penchat_submit_probe SHARED module.cpp ${SDK_SRC})
  target_include_directories(jsapi_penchat_submit_probe PRIVATE
    "${ROOT}/native/vendor/iot-miniapp-sdk/include" "${ROOT}/.deps/json/single_include")
  set_target_properties(jsapi_penchat_submit_probe PROPERTIES SKIP_BUILD_RPATH ON)
  target_compile_options(jsapi_penchat_submit_probe PRIVATE -include pthread.h)
  target_compile_definitions(jsapi_penchat_submit_probe PRIVATE PENCHAT_SUBMIT_PROBE_APP_ID="${PENCHAT_SUBMIT_PROBE_APP_ID}")
  target_link_libraries(jsapi_penchat_submit_probe PRIVATE submit_probe_core -static-libstdc++ -static-libgcc
    -Wl,-unresolved-symbols=ignore-all "-Wl,--version-script=${ROOT}/native/launcher/exports.map")
endif()
set_tests_properties(submit-probe composite-probe composite-surface PROPERTIES TIMEOUT 15)
# This host's GCC 12 sanitizer runtime sporadically faults before main for PIE,
# including an empty program (8/40); non-PIE empty programs passed 40/40.
# Keep shared objects PIC and all instrumentation enabled. Only test executables
# use a fixed ELF load address; neither the device nor production flags change.
if(NOT CMAKE_CROSSCOMPILING AND CMAKE_CXX_FLAGS MATCHES "fsanitize")
  foreach(test submit_probe_tests composite_tests composite_fixture surface_tests surface_fixture)
    target_link_options(${test} PRIVATE -no-pie)
  endforeach()
  # A raw SIGSEGV still fails the test; avoid recursive runtime signal reporting.
  set_tests_properties(submit-probe composite-probe composite-surface PROPERTIES ENVIRONMENT "ASAN_OPTIONS=handle_segv=0:abort_on_error=1")
endif()
