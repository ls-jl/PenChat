cmake_minimum_required(VERSION 3.20)
project(penchat_web_search LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EDGE_SEARCH_SOURCE "${ROOT}/.deps/edge-search" CACHE PATH "Verified and patched source")
set(EDGE_SEARCH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EDGE_SEARCH_USE_IMPERSONATE ON CACHE BOOL "" FORCE)
add_subdirectory("${EDGE_SEARCH_SOURCE}" edge EXCLUDE_FROM_ALL)
add_executable(penchat-web-search worker.cpp)
if(CMAKE_CROSSCOMPILING)
  target_sources(penchat-web-search PRIVATE sdk_runtime.cpp)
endif()
target_include_directories(penchat-web-search PRIVATE "${ROOT}/.deps/json/single_include")
target_link_libraries(penchat-web-search PRIVATE EdgeSearch::edgesearch pthread)
set_target_properties(penchat-web-search PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "$ORIGIN/../web-search/lib" INSTALL_RPATH "$ORIGIN/../web-search/lib")
