cmake_minimum_required(VERSION 3.16)
project(penchat_scanner LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SCANNER_TARGET "orange" CACHE STRING "Scanner hardware target")
option(SCANNER_TESTS "Build scanner contract and mock-process tests" ON)
option(SCANNER_BUILD_REPLAY "Build the non-production NPU replay harness" OFF)
option(SCANNER_DIAGNOSTIC_TRACE "Explicit opt-in one-scan raw recorder (development only)" OFF)
option(SCANNER_SANITIZE "Address/undefined behavior sanitizers" OFF)
set(SCANNER_CV_ROOT "" CACHE PATH "Pinned OpenCV root")
set(SCANNER_PNG_ROOT "" CACHE PATH "Pinned libpng root")
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/cmake/Identity.cmake")
if(CMAKE_CROSSCOMPILING)
  set(CMAKE_SKIP_BUILD_RPATH TRUE)
  set(CV_LIB "${SCANNER_CV_ROOT}/lib")
  set(PNG_LIB "${SCANNER_PNG_ROOT}/lib")
else()
  set(CV_LIB "${SCANNER_CV_ROOT}/lib/x86_64-linux-gnu")
  find_package(PNG 1.6 REQUIRED)
endif()
add_library(stream32 STATIC engine/src/stream.cpp engine/src/geometry.cpp engine/src/geometry_quality.cpp
  engine/src/geometry_decoder.cpp engine/src/canvas.cpp engine/src/cvi.cpp
  engine/src/resize_compat.cpp)
if(NOT SCANNER_TARGET STREQUAL "s6")
  target_sources(stream32 PRIVATE engine/src/rknn.cpp)
endif()
target_include_directories(stream32 PUBLIC engine/include PRIVATE engine/src engine/include/rknn)
target_include_directories(stream32 SYSTEM PUBLIC "${SCANNER_CV_ROOT}/include/opencv4" "${SCANNER_CV_ROOT}/include")
target_link_directories(stream32 PUBLIC "${CV_LIB}")
target_link_libraries(stream32 PUBLIC opencv_imgproc opencv_core pthread dl)
if(SCANNER_TARGET STREQUAL "s6")
  target_link_libraries(stream32 PUBLIC "${SCANNER_CV_ROOT}/share/OpenCV/3rdparty/lib/libtegra_hal.a" "${SCANNER_CV_ROOT}/share/OpenCV/3rdparty/lib/libzlib.a")
endif()
if(NOT CMAKE_CROSSCOMPILING)
  target_link_options(stream32 PUBLIC "LINKER:-rpath-link,${CV_LIB}"
    "LINKER:-rpath-link,${CV_LIB}/openblas-pthread")
endif()
target_compile_options(stream32 PRIVATE -Wall -Wextra -ffp-contract=off)
set(CMAKE_BUILD_RPATH "${CV_LIB}")
if(SCANNER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(stream32 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  add_executable(penchat-scanner src/main.cpp src/hardware.cpp src/png.cpp src/runtime_guard.cpp)
  if(SCANNER_DIAGNOSTIC_TRACE)
    target_compile_definitions(penchat-scanner PRIVATE SCANNER_DIAGNOSTIC_TRACE=1)
  endif()
  if(SCANNER_TARGET STREQUAL "s6")
    target_sources(penchat-scanner PRIVATE src/cvi_capture.cpp)
    target_compile_definitions(penchat-scanner PRIVATE PENCHAT_CVI_CAPTURE=1 JSON_HAS_CPP_14=1)
    target_link_options(penchat-scanner PRIVATE "LINKER:--dynamic-list=${CMAKE_CURRENT_SOURCE_DIR}/cvi-runtime-symbols.list" -static-libstdc++ -static-libgcc)
  else()
    target_link_options(penchat-scanner PRIVATE "LINKER:--dynamic-list=${CMAKE_CURRENT_SOURCE_DIR}/runtime-symbols.list")
  endif()
  set_property(TARGET penchat-scanner APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/runtime-symbols.list")
  target_include_directories(penchat-scanner PRIVATE src "${ROOT}/.deps/json/single_include")
  target_link_libraries(penchat-scanner PRIVATE stream32)
  if(CMAKE_CROSSCOMPILING)
    target_include_directories(penchat-scanner PRIVATE "${SCANNER_PNG_ROOT}/include")
    target_link_libraries(penchat-scanner PRIVATE "${PNG_LIB}/libpng16.so.16" "${PNG_LIB}/libz.so.1")
  else()
    target_link_libraries(penchat-scanner PRIVATE PNG::PNG)
    if(SCANNER_TESTS)
      target_compile_definitions(penchat-scanner PRIVATE SCANNER_TEST_SOURCE=1)
    endif()
  endif()
endif()
if(SCANNER_TESTS)
  enable_testing()
  add_executable(scanner-profile-tests tests/test_profile.cpp)
  target_include_directories(scanner-profile-tests PRIVATE src "${ROOT}/.deps/json/single_include")
  target_compile_options(scanner-profile-tests PRIVATE -UNDEBUG)
  add_test(NAME scanner-hardware-profiles COMMAND scanner-profile-tests)
  add_executable(scanner-sequence-gap-tests tests/test_sequence_gap.cpp)
  target_include_directories(scanner-sequence-gap-tests PRIVATE src)
  target_compile_options(scanner-sequence-gap-tests PRIVATE -UNDEBUG)
  add_test(NAME scanner-sequence-recovery COMMAND scanner-sequence-gap-tests)
  add_executable(scanner-runtime-io-tests tests/test_runtime_io.cpp)
  target_include_directories(scanner-runtime-io-tests PRIVATE src)
  target_compile_options(scanner-runtime-io-tests PRIVATE -UNDEBUG)
  add_test(NAME scanner-cache-sync-ranges COMMAND scanner-runtime-io-tests)
  add_library(scanner-guard-vendor SHARED tests/runtime_guard_vendor.cpp)
  add_executable(scanner-runtime-guard-tests tests/test_runtime_guard.cpp src/runtime_guard.cpp)
  target_include_directories(scanner-runtime-guard-tests PRIVATE src)
  target_compile_options(scanner-runtime-guard-tests PRIVATE -UNDEBUG)
  target_link_options(scanner-runtime-guard-tests PRIVATE "LINKER:--dynamic-list=${CMAKE_CURRENT_SOURCE_DIR}/runtime-symbols.list")
  target_link_libraries(scanner-runtime-guard-tests PRIVATE dl)
  add_test(NAME scanner-vendor-command-block COMMAND scanner-runtime-guard-tests $<TARGET_FILE:scanner-guard-vendor>)
  add_library(stream32_mock_cvi SHARED engine/tests/fake_cvi_runtime.cpp)
  target_include_directories(stream32_mock_cvi PRIVATE engine/include engine/src)
  add_executable(stream32-cvi-tests engine/tests/test_cvi_adapter.cpp)
  target_include_directories(stream32-cvi-tests PRIVATE engine/src)
  target_link_libraries(stream32-cvi-tests PRIVATE stream32)
  add_test(NAME scanner-cvi32-adapter COMMAND stream32-cvi-tests $<TARGET_FILE:stream32_mock_cvi>)
  add_library(stream32_mock_runtime SHARED engine/tests/fake_runtime.cpp)
  target_include_directories(stream32_mock_runtime PRIVATE engine/include/rknn)
  add_executable(stream32-adapter-tests engine/tests/test_adapter.cpp)
  target_link_libraries(stream32-adapter-tests PRIVATE stream32)
  add_test(NAME scanner-fixed32-adapter COMMAND stream32-adapter-tests $<TARGET_FILE:stream32_mock_runtime>)
  add_executable(stream32-contract-tests engine/tests/test_stream.cpp)
  target_include_directories(stream32-contract-tests PRIVATE engine/src)
  target_link_libraries(stream32-contract-tests PRIVATE stream32)
  add_test(NAME scanner-stream-contracts COMMAND stream32-contract-tests)
  set_tests_properties(scanner-stream-contracts PROPERTIES TIMEOUT 120)
  add_executable(scanner-geometry-quality-tests tests/test_geometry_quality.cpp)
  target_include_directories(scanner-geometry-quality-tests PRIVATE engine/src)
  target_link_libraries(scanner-geometry-quality-tests PRIVATE stream32)
  add_test(NAME scanner-geometry-quality COMMAND scanner-geometry-quality-tests)
  set_tests_properties(scanner-geometry-quality PROPERTIES TIMEOUT 60)
  add_executable(scanner-memory-tests tests/test_memory.cpp)
  target_include_directories(scanner-memory-tests PRIVATE engine/src)
  target_link_libraries(scanner-memory-tests PRIVATE stream32)
  add_test(NAME scanner-dynamic-memory COMMAND scanner-memory-tests)
  set_tests_properties(scanner-dynamic-memory PROPERTIES TIMEOUT 180)
  add_executable(scanner-selection-golden tests/selection_golden.cpp)
  target_include_directories(scanner-selection-golden PRIVATE engine/src)
  target_link_libraries(scanner-selection-golden PRIVATE stream32)
  add_test(NAME scanner-selection-golden COMMAND ${CMAKE_COMMAND}
    -DPROGRAM=$<TARGET_FILE:scanner-selection-golden>
    -DEXPECTED=${CMAKE_CURRENT_SOURCE_DIR}/tests/selection-golden.txt
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/compare_golden.cmake)
  set_tests_properties(scanner-selection-golden PROPERTIES TIMEOUT 120)
  if(NOT CMAKE_CROSSCOMPILING)
    add_executable(scanner-file-read-tests tests/test_file_read.cpp)
    target_include_directories(scanner-file-read-tests PRIVATE src)
    add_test(NAME scanner-process-file-read COMMAND scanner-file-read-tests)
    add_executable(scanner-preview-tests tests/test_preview.cpp)
    target_include_directories(scanner-preview-tests PRIVATE src)
    target_link_libraries(scanner-preview-tests PRIVATE pthread)
    add_test(NAME scanner-nonblocking-preview COMMAND scanner-preview-tests)
    set_tests_properties(scanner-nonblocking-preview PROPERTIES TIMEOUT 10)
    add_executable(scanner-png-tests tests/test_png.cpp src/png.cpp)
    target_include_directories(scanner-png-tests PRIVATE src)
    target_link_libraries(scanner-png-tests PRIVATE PNG::PNG)
    add_test(NAME scanner-row-png COMMAND scanner-png-tests)
    set_tests_properties(scanner-row-png PROPERTIES TIMEOUT 30)
  endif()
endif()

if(SCANNER_BUILD_REPLAY)
  add_executable(scanner-canvas-preservation EXCLUDE_FROM_ALL tests/test_canvas_preservation.cpp)
  target_include_directories(scanner-canvas-preservation PRIVATE engine/src)
  target_link_libraries(scanner-canvas-preservation PRIVATE stream32)
  add_executable(scanner-cvi-replay tests/cvi_replay.cpp)
  target_link_libraries(scanner-cvi-replay PRIVATE stream32)
  add_executable(scanner-replay tests/replay.cpp)
  target_include_directories(scanner-replay PRIVATE engine/src)
  target_link_libraries(scanner-replay PRIVATE stream32)
  if(SCANNER_TARGET STREQUAL "s6")
    target_link_options(scanner-canvas-preservation PRIVATE -static-libstdc++ -static-libgcc)
    target_compile_definitions(scanner-replay PRIVATE PENCHAT_REPLAY_CVI_ONLY=1)
    target_link_options(scanner-replay PRIVATE -static-libstdc++ -static-libgcc)
    target_link_options(scanner-cvi-replay PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()

if(SCANNER_TARGET STREQUAL "s6")
  add_executable(s6-capture-probe EXCLUDE_FROM_ALL "${ROOT}/tests/device/s6_capture_probe.cpp" src/hardware.cpp src/cvi_capture.cpp src/runtime_guard.cpp)
  target_include_directories(s6-capture-probe PRIVATE src "${ROOT}/.deps/json/single_include")
  target_compile_definitions(s6-capture-probe PRIVATE PENCHAT_CVI_CAPTURE=1 JSON_HAS_CPP_14=1)
  target_link_libraries(s6-capture-probe PRIVATE dl pthread)
  target_link_options(s6-capture-probe PRIVATE -static-libstdc++ -static-libgcc "LINKER:--dynamic-list=${CMAKE_CURRENT_SOURCE_DIR}/cvi-runtime-symbols.list")
endif()

if(SCANNER_TARGET STREQUAL "s6")
  add_library(penchat_cvi_media SHARED src/cvi_media_link.c)
  target_link_options(penchat_cvi_media PRIVATE -Wl,--no-as-needed -Wl,--allow-shlib-undefined)
  foreach(name sample vpu venc vdec isp isp_algo awb ae af sys sns_full ini cvitracer)
    target_link_libraries(penchat_cvi_media PRIVATE "${ROOT}/.deps/s6-media/lib${name}.so")
  endforeach()
  target_include_directories(penchat-scanner PRIVATE "${ROOT}/.deps/openssl-headers-s6/include")
  target_include_directories(s6-capture-probe PRIVATE "${ROOT}/.deps/openssl-headers-s6/include")
  target_link_libraries(penchat-scanner PRIVATE "${ROOT}/.deps/http-system-s6/lib/libcrypto.so.1.1")
  target_link_libraries(s6-capture-probe PRIVATE "${ROOT}/.deps/http-system-s6/lib/libcrypto.so.1.1")
endif()
