cmake_minimum_required(VERSION 3.16)
project(penchat_scanner LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SCANNER_TESTS "Build scanner contract and mock-process tests" ON)
option(SCANNER_BUILD_REPLAY "Build the non-production NPU replay harness" OFF)
option(SCANNER_SANITIZE "Address/undefined behavior sanitizers" OFF)
set(SCANNER_CV_ROOT "" CACHE PATH "Pinned OpenCV root")
set(SCANNER_PNG_ROOT "" CACHE PATH "Pinned libpng root")
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/cmake/Identity.cmake")
if(CMAKE_CROSSCOMPILING)
  set(CMAKE_SKIP_BUILD_RPATH TRUE)
  set(CV_LIB "${SCANNER_CV_ROOT}/lib")
  set(PNG_LIB "${SCANNER_PNG_ROOT}/lib")
else()
  set(CV_LIB "${SCANNER_CV_ROOT}/lib/x86_64-linux-gnu")
  find_package(PNG 1.6 REQUIRED)
endif()
add_library(stream32 STATIC engine/src/stream.cpp engine/src/geometry.cpp
  engine/src/geometry_decoder.cpp engine/src/canvas.cpp engine/src/rknn.cpp
  engine/src/resize_compat.cpp)
target_include_directories(stream32 PUBLIC engine/include PRIVATE engine/src engine/include/rknn)
target_include_directories(stream32 SYSTEM PUBLIC "${SCANNER_CV_ROOT}/include/opencv4")
target_link_directories(stream32 PUBLIC "${CV_LIB}")
target_link_libraries(stream32 PUBLIC opencv_imgproc opencv_core pthread dl)
if(NOT CMAKE_CROSSCOMPILING)
  target_link_options(stream32 PUBLIC "LINKER:-rpath-link,${CV_LIB}"
    "LINKER:-rpath-link,${CV_LIB}/openblas-pthread")
endif()
target_compile_options(stream32 PRIVATE -Wall -Wextra -ffp-contract=off)
set(CMAKE_BUILD_RPATH "${CV_LIB}")
if(SCANNER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(stream32 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  add_executable(penchat-scanner src/main.cpp src/hardware.cpp src/png.cpp)
  target_include_directories(penchat-scanner PRIVATE src "${ROOT}/.deps/json/single_include")
  target_link_libraries(penchat-scanner PRIVATE stream32)
  if(CMAKE_CROSSCOMPILING)
    target_include_directories(penchat-scanner PRIVATE "${SCANNER_PNG_ROOT}/include")
    target_link_libraries(penchat-scanner PRIVATE "${PNG_LIB}/libpng16.so.16" "${PNG_LIB}/libz.so.1")
  else()
    target_link_libraries(penchat-scanner PRIVATE PNG::PNG)
    if(SCANNER_TESTS)
      target_compile_definitions(penchat-scanner PRIVATE SCANNER_TEST_SOURCE=1)
    endif()
  endif()
endif()
if(SCANNER_TESTS)
  enable_testing()
  add_library(stream32_mock_runtime SHARED engine/tests/fake_runtime.cpp)
  target_include_directories(stream32_mock_runtime PRIVATE engine/include/rknn)
  add_executable(stream32-adapter-tests engine/tests/test_adapter.cpp)
  target_link_libraries(stream32-adapter-tests PRIVATE stream32)
  add_test(NAME scanner-fixed32-adapter COMMAND stream32-adapter-tests $<TARGET_FILE:stream32_mock_runtime>)
  add_executable(stream32-contract-tests engine/tests/test_stream.cpp)
  target_include_directories(stream32-contract-tests PRIVATE engine/src)
  target_link_libraries(stream32-contract-tests PRIVATE stream32)
  add_test(NAME scanner-stream-contracts COMMAND stream32-contract-tests)
  set_tests_properties(scanner-stream-contracts PROPERTIES TIMEOUT 120)
  if(NOT CMAKE_CROSSCOMPILING)
    add_executable(scanner-file-read-tests tests/test_file_read.cpp)
    target_include_directories(scanner-file-read-tests PRIVATE src)
    add_test(NAME scanner-process-file-read COMMAND scanner-file-read-tests)
    add_executable(scanner-preview-tests tests/test_preview.cpp)
    target_include_directories(scanner-preview-tests PRIVATE src)
    target_link_libraries(scanner-preview-tests PRIVATE pthread)
    add_test(NAME scanner-nonblocking-preview COMMAND scanner-preview-tests)
    set_tests_properties(scanner-nonblocking-preview PROPERTIES TIMEOUT 10)
    add_executable(scanner-png-tests tests/test_png.cpp src/png.cpp)
    target_include_directories(scanner-png-tests PRIVATE src)
    target_link_libraries(scanner-png-tests PRIVATE PNG::PNG)
    add_test(NAME scanner-row-png COMMAND scanner-png-tests)
    set_tests_properties(scanner-row-png PROPERTIES TIMEOUT 30)
  endif()
endif()

if(SCANNER_BUILD_REPLAY)
  add_executable(scanner-replay tests/replay.cpp)
  target_include_directories(scanner-replay PRIVATE engine/src)
  target_link_libraries(scanner-replay PRIVATE stream32)
endif()
